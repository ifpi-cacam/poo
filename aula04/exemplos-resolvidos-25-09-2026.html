Conta de Energia

Problema

Representar a conta mensal de energia de um cliente. O objeto deve armazenar número da instalação, cliente, consumo em kWh e valor do kWh.

O consumo só pode ser atualizado por método público e não aceita valor negativo.

classDiagram

        -int numeroInstalacao
        -String nomeCliente
        -double consumoKwh
        -double valorKwh
        +ContaEnergia(numero, cliente, consumo, valor)
        +registrarConsumo(kwh)
        +calcularValor()
        +exibirResumo()
    }
ContaEnergia.java
package contaenergia;

public class ContaEnergia {

    private int numeroInstalacao;
    private String nomeCliente;
    private double consumoKwh;
    private double valorKwh;

    public ContaEnergia(int numeroInstalacao,
                        String nomeCliente,
                        double consumoKwh,
                        double valorKwh) {
        this.numeroInstalacao = numeroInstalacao;
        this.nomeCliente = nomeCliente;
        this.consumoKwh = consumoKwh;
        this.valorKwh = valorKwh;
    }

    public void registrarConsumo(double kwh) {
        if (kwh > 0) {
            consumoKwh = kwh;
        }
    }

    public double calcularValor() {
        return valorKwh * consumoKwh;
    }

    public void exibirResumo() {
        System.out.println("Resumo do consumo de energia");
        System.out.println("Numero da Instalacao: " + numeroInstalacao);
        System.out.println("Nome do Cliente: " + nomeCliente);
        System.out.println("Valor do KwH: " + valorKwh);
        System.out.println("Consumo de Kw: " + consumoKwh);
        System.out.println("Valor total consumido: " + calcularValor());
    }
}

    }
}

Produto em Estoque

Problema

Representar um produto com código, nome, preço e quantidade em estoque.

O programa principal deve criar objetos, repor estoque, realizar uma venda válida, tentar uma venda maior que o estoque, alterar preço e exibir o estado final.

classDiagram

        ~int codigo
        ~String nome
        ~double preco
        ~int quantidadeEstoque
        +Produto(codigo, nome, preco, estoque)
        +repor(quantidade)
        +vender(quantidade)
        +alterarPreco(novoPreco)
        +exibirDados()
    }
Produto.java
package produtoestoque;

public class Produto {
    int codigo;
    String nome;
    double preco;
    int quantidadeEstoque;

    public Produto(int codigo, String nome,
                   double preco, int quantidadeEstoque) {
        this.codigo = codigo;
        this.nome = nome;
        this.preco = preco;
        this.quantidadeEstoque = quantidadeEstoque;
    }

    public void repor(int quantidade) {
        if (quantidade >= 0) {
            quantidadeEstoque = quantidade;
        }
    }

    public boolean vender(int quantidade) {
        if (quantidade < 0) {
            return false;
        }

        if (quantidadeEstoque >= quantidade) {
            quantidadeEstoque = quantidadeEstoque - quantidade;
            return true;
        }
        return false;
    }

    public void alterarPreco(double novoPreco) {
        if (novoPreco >= 0) {
            preco = novoPreco;
        }
    }

    public void exibirDados() {
        System.out.println("Dados do Produto");
        System.out.println("Codigo: " + codigo);
        System.out.println("Nome: " + nome);
        System.out.println("Preco: " + preco);
        System.out.println("Quantidade: " + quantidadeEstoque);
    }
}
ProgramaProduto.java
package produtoestoque;

public class ProgramaProduto {

    public static void main(String[] args) {
        Produto mouse = new Produto(101, "Mouse USB", 45.90, 10);
        Produto teclado = new Produto(102, "Teclado ABNT2", 89.90, 5);

        mouse.repor(15);
        teclado.repor(8);

        boolean vendaMouse = mouse.vender(3);
        boolean vendaTeclado = teclado.vender(20);

        mouse.alterarPreco(42.50);
        teclado.alterarPreco(84.90);

        System.out.println("Venda valida do mouse realizada? " + vendaMouse);
        System.out.println("Venda maior que o estoque do teclado realizada? " + vendaTeclado);
        System.out.println();

        mouse.exibirDados();
        System.out.println();
        teclado.exibirDados();
    }
}

Conta Bancária

Problema

Representar uma conta bancária com titular e saldo privado.

O saldo só pode mudar por depósito ou saque. O programa principal cria duas contas e consulta o estado por métodos públicos.

classDiagram

        +sacar(valor)
        +getSaldo()
        +getTitular()
    }
Conta.java
package ifpi.contabancaria;

public class Conta {
    private String titular;
    private double saldo;

    public Conta(String t, double s) {
        saldo = s;
        titular = t;
    }

    public void depositar(double valor) {
        if (valor > 0) {
            saldo += valor;
        }
    }

    public boolean sacar(double valor) {
        if (valor > 0 && valor <= saldo) {
            saldo -= valor;
            return true;
        }
        return false;
    }

    public double getSaldo() {
        return saldo;
    }

    public String getTitular() {
        return titular;
    }
}
Programa.java
package ifpi.contabancaria;

public class Programa {
    public static void main(String[] args) {
        Conta c1 = new Conta("Ana", 100);
        Conta c2 = new Conta("Paulo", 500);

        System.out.println("Saldo da conta c1: " + c1.getSaldo());
        System.out.println("Titular da conta c1: " + c1.getTitular());
        System.out.println("Saldo da conta c2: " + c2.getSaldo());
        System.out.println("Titular da conta c2: " + c2.getTitular());

        c1.depositar(1000);
        c1.sacar(250);

        c2.depositar(400);

        System.out.println(c1.getSaldo());
        System.out.println(c2.getSaldo());
    }
}
